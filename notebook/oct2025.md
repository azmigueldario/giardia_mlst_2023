### 2025-Oct-28

All sequences were downloaded in gene space from NCBI datasets using `curl` into a subfolder of the `/project` mount

```sh
#################################### Download data ####################################

curl -X GET "https://api.ncbi.nlm.nih.gov/datasets/v2/gene/id/5698276,5699725,5700156,5697398,5699357,5697828,5701000,5699927,5700811/download?include_annotation_type=FASTA_PROTEIN&include_annotation_type=FASTA_GENE" \
 -H 'accept: application/zip' \
 --output MLST.zip

# clean names and split fasta
seqkit replace \
    --pattern "\.1:[c0-9]+-[0-9]+ " \
    --replacement "-1:" \
    gene.fna |
seqkit split \
    --by-id \
    --line-width 80  \
    --id-regexp "(^.*GL[0-9_]+) " \
    --two-pass 

# rename output
rename "stdin.part_" "" *
```

### 2025-Oct-29

Bumped version of chewbbaca to 3.4.2, will have to integrate into pipeline and fix any issues. This new version will let me get the fasta sequences for a MSA and a phylogenetic tree more easily as the functionality was integrated into the tool. 

```sh
#################################### MLST detection and MSA ####################################
# interactive session for development
salloc --time=4:00:00 --ntasks=1 --cpus-per-task=8  --mem=80G

# software requirements
module load StdEnv/2023 apptainer/1.3.5

#  ENV variables for paths and input samplesheet
input_filepaths_list="/project/60006/mdprieto/giardia_mlst_2023/scripts/nf_chewbbaca/fofn_clustered_HQ.txt"
mlst_fasta_dir="/project/60006/mdprieto/raw_data/giardia/MLST_loci"
results_dir="/scratch/mdprieto/results/giardia_classic_mlst"

# ENV variables for containers and executable scripts
chewbbaca_img="/scratch/group_share/singularity_imgs/depot.galaxyproject.org-singularity-chewbbaca-3.4.2-pyhdfd78af_0.sif"
iqtree_img="/scratch/group_share/singularity_imgs/depot.galaxyproject.org-singularity-iqtree-2.3.4--h21ec9f0_0.img"
makemergetable_rb="/project/60006/mdprieto/giardia_mlst_2023/scripts/classic_schemas/makemergetable.rb"


apptainer exec ${chewbbaca_img} chewBBACA.py PrepExternalSchema \
    --schema-directory "${mlst_fasta_dir}/classic_three_loci/" \
    --translation-table 1 \
    --output-directory "${results_dir}/three_loci/adapted_schema" \
    --cpu-cores 6

apptainer exec ${chewbbaca_img} chewBBACA.py AlleleCall \
    --input-files ${input_filepaths_list} \
    --schema-directory "${results_dir}/three_loci/adapted_schema" \
    --translation-table 1 \
    --output-directory ${results_dir}/three_loci/AlleleCall \
    --cpu-cores 6

apptainer exec ${chewbbaca_img} chewBBACA.py GetAlleles \
    --input-file ${results_dir}/three_loci/call/results_alleles.tsv  \
    --schema-directory "${results_dir}/three_loci/adapted_schema" \
    --translation-table 1 \
    --output-directory ${results_dir}/three_loci/GetAlleles \
    --cpu-cores 6

mkdir -p ${results_dir}/three_loci/multiple_alignment &&
    cd ${results_dir}/three_loci/GetAlleles/fastas &&
    for fasta_file in $(ls *.fasta)
    do
        apptainer exec ${chewbbaca_img} mafft \
            --localpair \
            --maxiterate 1000 \
            "${fasta_file}" > "${results_dir}/three_loci/multiple_alignment/${fasta_file%.fasta}_aligned.fasta"
    done


cd ${results_dir}/three_loci/multiple_alignment &&
    ruby ${makemergetable_rb} *.fasta > subMSAtable &&
    cat *.fasta > three_loci_concatenated.fasta &&
    apptainer exec ${chewbbaca_img} mafft \
        --localpair \
        --maxiterate 100 \
        --merge \
        subMSAtable three_loci_concatenated.fasta > merged_MSA_three_loci.fasta


```
